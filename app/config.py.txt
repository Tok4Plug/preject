+def _env_bool(name: str, default: bool) -> bool:
+    value = os.getenv(name)
+    if value is None:
+        return default
+    return value.strip().lower() in {"1", "true", "yes", "on"}
+
+
+def _env_int(name: str, default: int) -> int:
+    value = os.getenv(name)
+    if value is None:
+        return default
+    try:
+        return int(value)
+    except ValueError:
+        logger.warning("Valor inválido para %s=%r. Usando default=%s", name, value, default)
+        return default
+
+
+def _env_json_dict(name: str, default: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
+    default = default or {}
+    raw_value = os.getenv(name)
+    if not raw_value:
+        return default
+
+    try:
+        parsed = json.loads(raw_value)
+    except json.JSONDecodeError:
+        logger.warning("JSON inválido em %s. Usando default.", name)
+        return default
+
+    if not isinstance(parsed, dict):
+        logger.warning("%s deve ser objeto JSON. Usando default.", name)
+        return default
+
+    return parsed
