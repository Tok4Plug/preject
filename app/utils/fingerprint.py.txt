@dataclass(frozen=True)
class FingerprintComponents:
    """
    Componentes imutáveis para geração de fingerprint.
    A ordem dos campos importa para o hash consistente.
    """
    # Identificadores principais
    telegram_id: Optional[str] = None
    telegram_username: Optional[str] = None
    phone_number_hash: Optional[str] = None
    
    # Dados de sessão/conexão
    session_id: Optional[str] = None
    ip_address: Optional[str] = None
    user_agent_hash: Optional[str] = None
    language_code: Optional[str] = None
    
    # Contexto da aplicação
    tenant_id: Optional[str] = None
    bot_id: Optional[str] = None
    flow_id: Optional[str] = None
    
    # Metadados
    created_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    version: str = "1.0"
    
    def __post_init__(self):
        """Validação pós-inicialização."""
        # Normaliza campos string
        if self.telegram_username:
            object.__setattr__(self, 'telegram_username', self.telegram_username.lower().strip())
        
        if self.language_code:
            object.__setattr__(self, 'language_code', self.language_code.lower().strip())
        
        # Validações básicas
        if self.telegram_id and not re.match(r'^\d+$', str(self.telegram_id)):
            raise ValidationError(f"Telegram ID inválido: {self.telegram_id}")
        
        if self.tenant_id and not self.tenant_id.startswith('tenant_'):
            raise ValidationError(f"Tenant ID deve começar com 'tenant_': {self.tenant_id}")
    
    def to_dict(self) -> Dict[str, Any]:
        """Converte para dict ordenado para hash consistente."""
        return asdict(self)
    
    def to_json_string(self) -> str:
        """Serializa para JSON com ordenação de chaves."""
        return json.dumps(self.to_dict(), sort_keys=True, separators=(',', ':'))
    
    def get_primary_identifier(self) -> str:
        """Retorna o identificador principal mais confiável."""
        if self.telegram_id:
            return f"telegram:{self.telegram_id}"
        elif self.phone_number_hash:
            return f"phone:{self.phone_number_hash}"
        elif self.session_id:
            return f"session:{self.session_id}"
        else:
            raise FingerprintError("Nenhum identificador primário disponível")


class FingerprintEngine:
    """Motor de geração e validação de fingerprints."""
    
    # Versões de algoritmo suportadas
    ALGORITHM_VERSIONS = {
        "1.0": hashlib.sha256,
        "1.1": hashlib.sha384,  # Futura migração
    }
    
    # Componentes obrigatórios por contexto
    REQUIRED_COMPONENTS = {
        "anti_fraud": ["telegram_id", "session_id", "ip_address"],
        "tracking": ["telegram_id", "bot_id", "flow_id"],
        "payment": ["telegram_id", "tenant_id", "phone_number_hash"],
        "access": ["telegram_id", "tenant_id", "session_id"],
    }
    
    def __init__(self, salt: Optional[str] = None):
        """
        Inicializa o motor de fingerprint.
        
        Args:
            salt: Salt para hashing (opcional, busca de ENV)
        """
        self.salt = salt or self._load_salt()
        self.version = "1.0"
        self.hash_algorithm = self.ALGORITHM_VERSIONS[self.version]
    
    def _load_salt(self) -> str:
        """Carrega salt de ambiente ou gera um."""
        salt = hashlib.sha256(
            os.urandom(32) + str(uuid.uuid4()).encode()
        ).hexdigest()
        
        # Em produção, use uma variável de ambiente
        env_salt = os.getenv('FINGERPRINT_SALT')
        if env_salt:
            salt = env_salt
        
        return salt
    
    def generate(self, components: FingerprintComponents, 
                context: str = "default") -> str:
        """
        Gera fingerprint único baseado em componentes.
        
        Args:
            components: Componentes do fingerprint
            context: Contexto de uso (anti_fraud, tracking, etc.)
            
        Returns:
            Fingerprint hash
        
        Raises:
            FingerprintError: Se falhar na geração
        """
        try:
            # Valida componentes obrigatórios para o contexto
            if context in self.REQUIRED_COMPONENTS:
                required = self.REQUIRED_COMPONENTS[context]
                missing = [field for field in required 
                          if not getattr(components, field, None)]
                
                if missing:
                    raise FingerprintError(
                        f"Componentes obrigatórios faltando para contexto '{context}': "
                        f"{missing}"
                    )
            
            # Serializa componentes de forma consistente
            data_string = components.to_json_string()
            
            # Adiciona salt e contexto
            salted_data = f"{self.salt}:{context}:{data_string}"
            
            # Gera hash
            hash_obj = self.hash_algorithm()
            hash_obj.update(salted_data.encode('utf-8'))
            fingerprint = hash_obj.hexdigest()
            
            # Registra auditoria
            logger.info(
                f"Fingerprint gerado: context={context}, "
                f"primary_id={components.get_primary_identifier()}, "
                f"hash={fingerprint[:16]}..."
            )
            
            return fingerprint
            
        except Exception as e:
            logger.error(f"Falha ao gerar fingerprint: {str(e)}")
            raise FingerprintError(f"Erro na geração de fingerprint: {str(e)}")
    
    def generate_from_dict(self, data: Dict[str, Any], 
                          context: str = "default") -> str:
        """
        Gera fingerprint a partir de dict.
        
        Args:
            data: Dados para fingerprint
            context: Contexto de uso
            
        Returns:
            Fingerprint hash
        """
        # Filtra e converte campos
        components_data = {}
        
        field_mapping = {
            'user_id': 'telegram_id',
            'username': 'telegram_username',
            'phone': 'phone_number_hash',
            'session': 'session_id',
            'ip': 'ip_address',
            'user_agent': 'user_agent_hash',
            'lang': 'language_code',
            'tenant': 'tenant_id',
            'bot': 'bot_id',
            'flow': 'flow_id',
        }
        
        # Mapeia campos
        for source_field, target_field in field_mapping.items():
            if source_field in data:
                components_data[target_field] = data[source_field]
        
        # Campos diretos (se já estiverem no formato correto)
        for field in FingerprintComponents.__annotations__.keys():
            if field in data and field not in components_data:
                components_data[field] = data[field]
        
        # Processa campos especiais
        if 'phone_number' in data and not components_data.get('phone_number_hash'):
            # Hash do número de telefone (mantém privacidade)
            phone = str(data['phone_number']).strip()
            if phone:
                phone_hash = hashlib.sha256(
                    f"{self.salt}:phone:{phone}".encode()
                ).hexdigest()
                components_data['phone_number_hash'] = phone_hash
        
        if 'user_agent' in data and not components_data.get('user_agent_hash'):
            # Hash do user agent
            ua = str(data['user_agent']).strip()
            if ua:
                ua_hash = hashlib.sha256(
                    f"{self.salt}:ua:{ua}".encode()
                ).hexdigest()
                components_data['user_agent_hash'] = ua_hash
        
        # Cria componentes
        components = FingerprintComponents(**components_data)
        
        return self.generate(components, context)
    
    def validate(self, fingerprint: str, components: FingerprintComponents,
                context: str = "default") -> bool:
        """
        Valida fingerprint em relação aos componentes.
        
        Args:
            fingerprint: Fingerprint a validar
            components: Componentes esperados
            context: Contexto original
            
        Returns:
            True se válido
            
        Raises:
            FingerprintError: Se inválido
        """
        try:
            # Regera fingerprint dos componentes
            expected = self.generate(components, context)
            
            # Comparação segura
            if not hmac.compare_digest(fingerprint, expected):
                raise FingerprintError("Fingerprint não corresponde aos componentes")
            
            return True
            
        except Exception as e:
            logger.error(f"Validação de fingerprint falhou: {str(e)}")
            raise FingerprintError(f"Fingerprint inválido: {str(e)}")
    
    def get_fingerprint_tier(self, fingerprint: str) -> str:
        """
        Classifica a qualidade/confiança do fingerprint.
        
        Args:
            fingerprint: Fingerprint a classificar
            
        Returns:
            Tier: 'high', 'medium', 'low'
        """
        # Em uma implementação real, analisaria:
        # - Quantidade de componentes
        # - Confiabilidade dos componentes
        # - Histórico de uso
        
        # Por enquanto, retorna baseado no prefixo do hash
        # (implementação simplificada)
        first_char = fingerprint[0].lower()
        
        if first_char in '01234567':
            return 'high'
        elif first_char in '89abcdef':
            return 'medium'
        else:
            return 'low'
    
    def create_composite_fingerprint(self, fingerprints: List[str]) -> str:
        """
        Cria fingerprint composto de múltiplos fingerprints.
        
        Args:
            fingerprints: Lista de fingerprints
            
        Returns:
            Fingerprint composto
        """
        if not fingerprints:
            raise FingerprintError("Lista de fingerprints vazia")
        
        # Ordena para consistência
        sorted_fingerprints = sorted(fingerprints)
        
        # Combina com delimitador único
        combined = "|".join(sorted_fingerprints)
        
        # Hash da combinação
        hash_obj = self.hash_algorithm()
        hash_obj.update(f"{self.salt}:composite:{combined}".encode())
        
        return hash_obj.hexdigest()
    
    def generate_session_fingerprint(self, user_data: Dict[str, Any]) -> Tuple[str, str]:
        """
        Gera fingerprint de sessão com ID de sessão.
        
        Args:
            user_data: Dados do usuário
            
        Returns:
            Tupla (session_id, session_fingerprint)
        """
        # Gera ID de sessão único
        session_id = f"sess_{uuid.uuid4().hex}"
        
        # Adiciona session_id aos componentes
        components_data = user_data.copy()
        components_data['session_id'] = session_id
        
        # Gera fingerprint
        fingerprint = self.generate_from_dict(components_data, context="session")
        
        return session_id, fingerprint
    
    def detect_similarity(self, fp1: str, fp2: str, threshold: float = 0.8) -> float:
        """
        Detecta similaridade entre fingerprints usando distância de Hamming.
        
        Args:
            fp1: Primeiro fingerprint
            fp2: Segundo fingerprint
            threshold: Limite de similaridade (0-1)
            
        Returns:
            Score de similaridade (0-1)
        """
        # Converte para bytes para comparação
        try:
            bytes1 = bytes.fromhex(fp1)
            bytes2 = bytes.fromhex(fp2)
        except ValueError:
            # Se não for hex, usa string
            bytes1 = fp1.encode()
            bytes2 = fp2.encode()
        
        # Garante mesmo comprimento
        min_len = min(len(bytes1), len(bytes2))
        bytes1 = bytes1[:min_len]
        bytes2 = bytes2[:min_len]
        
        # Calcula similaridade
        if min_len == 0:
            return 0.0
        
        # Conta bytes iguais
        equal_count = sum(b1 == b2 for b1, b2 in zip(bytes1, bytes2))
        similarity = equal_count / min_len
        
        return similarity
    
    def is_high_risk_fingerprint(self, fingerprint: str, 
                                user_history: Dict[str, Any]) -> bool:
        """
        Avalia se fingerprint é de alto risco.
        
        Args:
            fingerprint: Fingerprint a avaliar
            user_history: Histórico do usuário
            
        Returns:
            True se for considerado alto risco
        """
        # Implementação básica - expandir com regras de negócio
        rules = [
            # Muitas sessões em pouco tempo
            lambda: len(user_history.get('recent_sessions', [])) > 5,
            
            # Mudança frequente de IP
            lambda: len(set(user_history.get('ip_history', []))) > 3,
            
            # Fingerprint muito recente (possível fraude)
            lambda: user_history.get('fingerprint_age_hours', 999) < 1,
            
            # Similaridade com fingerprints bloqueados
            lambda: any(
                self.detect_similarity(fingerprint, blocked_fp) > 0.9
                for blocked_fp in user_history.get('blocked_fingerprints', [])
            ),
        ]
        
        # Avalia regras
        risk_score = sum(1 for rule in rules if rule())
        
        return risk_score >= 2  # Duas ou mais regras acionadas


class FingerprintRegistry:
    """Registro e cache de fingerprints para análise e anti-fraude."""
    
    def __init__(self, redis_client=None):
        """
        Inicializa o registro.
        
        Args:
            redis_client: Cliente Redis para cache distribuído
        """
        self.redis = redis_client
        self.engine = FingerprintEngine()
        self.prefix = "fp:"
        
        # Cache local para performance
        self._local_cache = {}
        self._cache_ttl = 300  # 5 minutos
        
    def register(self, fingerprint: str, metadata: Dict[str, Any]) -> None:
        """
        Registra fingerprint no sistema.
        
        Args:
            fingerprint: Fingerprint a registrar
            metadata: Metadados associados
        """
        try:
            key = f"{self.prefix}{fingerprint}"
            data = {
                'fingerprint': fingerprint,
                'first_seen': datetime.now(timezone.utc).isoformat(),
                'last_seen': datetime.now(timezone.utc).isoformat(),
                'metadata': metadata,
                'count': 1
            }
            
            # Redis distribuído
            if self.redis:
                pipe = self.redis.pipeline()
                pipe.hset(key, mapping=data)
                pipe.expire(key, 86400 * 7)  # 7 dias
                pipe.execute()
            
            # Cache local
            self._local_cache[fingerprint] = {
                'data': data,
                'expires': datetime.now(timezone.utc).timestamp() + self._cache_ttl
            }
            
            logger.info(f"Fingerprint registrado: {fingerprint[:16]}...")
            
        except Exception as e:
            logger.error(f"Falha ao registrar fingerprint: {str(e)}")
    
    def update(self, fingerprint: str, metadata: Dict[str, Any]) -> None:
        """
        Atualiza registro de fingerprint.
        
        Args:
            fingerprint: Fingerprint a atualizar
            metadata: Novos metadados
        """
        try:
            existing = self.get(fingerprint)
            if not existing:
                self.register(fingerprint, metadata)
                return
            
            # Atualiza dados
            existing['last_seen'] = datetime.now(timezone.utc).isoformat()
            existing['count'] = existing.get('count', 0) + 1
            
            # Merge metadata
            if 'metadata' in existing:
                existing['metadata'].update(metadata)
            else:
                existing['metadata'] = metadata
            
            # Salva
            key = f"{self.prefix}{fingerprint}"
            if self.redis:
                self.redis.hset(key, mapping=existing)
                self.redis.expire(key, 86400 * 7)
            
            # Atualiza cache local
            self._local_cache[fingerprint] = {
                'data': existing,
                'expires': datetime.now(timezone.utc).timestamp() + self._cache_ttl
            }
            
        except Exception as e:
            logger.error(f"Falha ao atualizar fingerprint: {str(e)}")
    
    def get(self, fingerprint: str) -> Optional[Dict[str, Any]]:
        """
        Obtém registro de fingerprint.
        
        Args:
            fingerprint: Fingerprint a buscar
            
        Returns:
            Dados do fingerprint ou None
        """
        # Verifica cache local primeiro
        cache_entry = self._local_cache.get(fingerprint)
        if cache_entry and cache_entry['expires'] > datetime.now(timezone.utc).timestamp():
            return cache_entry['data'].copy()
        
        # Busca no Redis
        if self.redis:
            try:
                key = f"{self.prefix}{fingerprint}"
                data = self.redis.hgetall(key)
                if data:
                    # Converte bytes para string se necessário
                    decoded_data = {}
                    for k, v in data.items():
                        if isinstance(k, bytes):
                            k = k.decode('utf-8')
                        if isinstance(v, bytes):
                            v = v.decode('utf-8')
                        decoded_data[k] = v
                    
                    # Atualiza cache local
                    self._local_cache[fingerprint] = {
                        'data': decoded_data,
                        'expires': datetime.now(timezone.utc).timestamp() + self._cache_ttl
                    }
                    
                    return decoded_data.copy()
            except Exception as e:
                logger.error(f"Erro ao buscar fingerprint no Redis: {str(e)}")
        
        return None
    
    def find_similar(self, fingerprint: str, threshold: float = 0.7) -> List[Dict[str, Any]]:
        """
        Encontra fingerprints similares.
        
        Args:
            fingerprint: Fingerprint de referência
            threshold: Limite de similaridade
            
        Returns:
            Lista de fingerprints similares com metadados
        """
        # Nota: Em produção, use Redisearch ou outro mecanismo de busca
        # Esta é uma implementação simplificada
        
        similar = []
        
        # Busca no cache local
        for cached_fp, cache_entry in self._local_cache.items():
            if cache_entry['expires'] > datetime.now(timezone.utc).timestamp():
                similarity = self.engine.detect_similarity(fingerprint, cached_fp)
                if similarity >= threshold:
                    similar.append({
                        'fingerprint': cached_fp,
                        'similarity': similarity,
                        'data': cache_entry['data']
                    })
        
        return sorted(similar, key=lambda x: x['similarity'], reverse=True)
    
    def get_user_fingerprints(self, user_identifier: str) -> List[str]:
        """
        Obtém todos os fingerprints associados a um usuário.
        
        Args:
            user_identifier: ID do usuário (telegram_id, etc.)
            
        Returns:
            Lista de fingerprints
        """
        # Em produção, mantenha um índice de user->fingerprints
        # Esta é uma implementação simplificada
        
        user_fingerprints = []
        user_prefix = f"user:{user_identifier}:"
        
        if self.redis:
            try:
                # Busca por padrão (simplificado)
                keys = self.redis.keys(f"{self.prefix}*")
                for key in keys:
                    data = self.redis.hgetall(key)
                    if data.get(b'user_identifier', b'').decode() == user_identifier:
                        fingerprint = key.decode().replace(self.prefix, '')
                        user_fingerprints.append(fingerprint)
            except Exception as e:
                logger.error(f"Erro ao buscar fingerprints do usuário: {str(e)}")
        
        return user_fingerprints
    
    def is_blocked(self, fingerprint: str) -> bool:
        """
        Verifica se fingerprint está bloqueado.
        
        Args:
            fingerprint: Fingerprint a verificar
            
        Returns:
            True se bloqueado
        """
        block_key = f"blocked:{fingerprint}"
        
        if self.redis:
            try:
                return bool(self.redis.exists(block_key))
            except Exception as e:
                logger.error(f"Erro ao verificar fingerprint bloqueado: {str(e)}")
        
        return False
    
    def block(self, fingerprint: str, reason: str, 
              expires_hours: int = 24) -> None:
        """
        Bloqueia fingerprint.
        
        Args:
            fingerprint: Fingerprint a bloquear
            reason: Razão do bloqueio
            expires_hours: Expiração em horas
        """
        block_key = f"blocked:{fingerprint}"
        block_data = {
            'fingerprint': fingerprint,
            'blocked_at': datetime.now(timezone.utc).isoformat(),
            'reason': reason,
            'expires_at': (datetime.now(timezone.utc) + 
                          timedelta(hours=expires_hours)).isoformat()
        }
        
        if self.redis:
            try:
                self.redis.hset(block_key, mapping=block_data)
                self.redis.expire(block_key, expires_hours * 3600)
                logger.warning(f"Fingerprint bloqueado: {fingerprint[:16]}..., reason: {reason}")
            except Exception as e:
                logger.error(f"Erro ao bloquear fingerprint: {str(e)}")

    Valida fingerprint do usuário.

def register_fingerprint_event(fingerprint: str, event_type: str, 
                              metadata: Optional[Dict] = None) -> None:
    """
    Registra evento associado a fingerprint.
    
    Args:
        fingerprint: Fingerprint
        event_type: Tipo de evento (login, payment, etc.)
        metadata: Metadados do evento
    """
    if metadata is None:
        metadata = {}
    
    event_data = {
        'event_type': event_type,
        'timestamp': datetime.now(timezone.utc).isoformat(),
        **metadata
    }
    
    fingerprint_registry.update(fingerprint, {'last_event': event_data})


def check_fingerprint_risk(fingerprint: str, 
                          user_data: Dict[str, Any]) -> Dict[str, Any]:
    """
    Avalia risco de fingerprint.
    
    Args:
        fingerprint: Fingerprint a avaliar
        user_data: Dados do usuário
    
    Returns:
        Dict com análise de risco
    """
    registry_data = fingerprint_registry.get(fingerprint) or {}
    user_history = registry_data.get('metadata', {})
    
    # Análise de risco
    is_blocked = fingerprint_registry.is_blocked(fingerprint)
    tier = fingerprint_engine.get_fingerprint_tier(fingerprint)
    is_high_risk = fingerprint_engine.is_high_risk_fingerprint(fingerprint, user_history)
    
    # Histórico de uso
    first_seen = registry_data.get('first_seen')
    last_seen = registry_data.get('last_seen')
    usage_count = registry_data.get('count', 0)
    
    # Calcula idade
    age_hours = 0
    if first_seen:
        first_seen_dt = datetime.fromisoformat(first_seen.replace('Z', '+00:00'))
        age_hours = (datetime.now(timezone.utc) - first_seen_dt).total_seconds() / 3600
    
    # Score de risco (0-100)
    risk_score = 0
    
    if is_blocked:
        risk_score = 100
    elif is_high_risk:
        risk_score = 80
    elif tier == 'low':
        risk_score += 20
    elif age_hours < 1:
        risk_score += 30
    elif usage_count > 10:
        risk_score -= 10  # Fingerprint estabelecido
    
    risk_score = max(0, min(100, risk_score))
    
    return {
        'fingerprint': fingerprint,
        'is_blocked': is_blocked,
        'risk_score': risk_score,
        'risk_level': 'high' if risk_score >= 70 else 
                     'medium' if risk_score >= 30 else 'low',
        'tier': tier,
        'age_hours': age_hours,
        'usage_count': usage_count,
        'first_seen': first_seen,
        'last_seen': last_seen,
        'recommendation': 'block' if risk_score >= 80 else 
                         'review' if risk_score >= 50 else 'allow'
    }
