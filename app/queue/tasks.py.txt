class TaskPriority(Enum):
    """Prioridades de processamento de tarefas"""
    CRITICAL = 0    # Processamento imediato (webhooks, pagamentos)
    HIGH = 1        # Processamento rápido (notificações, antifraude)
    NORMAL = 2      # Processamento padrão (relatórios, analytics)
    LOW = 3         # Processamento em background (limpeza, manutenção)
    BATCH = 4       # Processamento em lote (syncs massivos)


class TaskStatus(Enum):
    """Status do processamento de uma tarefa"""
    PENDING = "pending"
    PROCESSING = "processing"
    COMPLETED = "completed"
    FAILED = "failed"
    RETRYING = "retrying"
    CANCELLED = "cancelled"


@dataclass(frozen=True)
class TaskMetadata:
    """Metadados imutáveis para uma tarefa"""
    task_id: UUID = field(default_factory=uuid4)
    tenant_id: str
    correlation_id: str = field(default_factory=lambda: str(uuid4()))
    created_at: datetime = field(default_factory=datetime.utcnow)
    scheduled_for: Optional[datetime] = None
    priority: TaskPriority = TaskPriority.NORMAL
    source: str = "unknown"
    parent_task_id: Optional[UUID] = None
    expires_at: Optional[datetime] = None
    
    def to_dict(self) -> Dict[str, Any]:
        """Converte metadados para dicionário serializável"""
        return {
            "task_id": str(self.task_id),
            "tenant_id": self.tenant_id,
            "correlation_id": self.correlation_id,
            "created_at": self.created_at.isoformat(),
            "scheduled_for": self.scheduled_for.isoformat() if self.scheduled_for else None,
            "priority": self.priority.value,
            "source": self.source,
            "parent_task_id": str(self.parent_task_id) if self.parent_task_id else None,
            "expires_at": self.expires_at.isoformat() if self.expires_at else None
        }


@dataclass(frozen=True)
class BaseTask:
    """Classe base para todas as tarefas"""
    metadata: TaskMetadata
    task_type: str
    payload: Dict[str, Any]
    idempotency_key: Optional[str] = None
    max_retries: int = 3
    retry_delay: int = 60  # segundos
    
    def __post_init__(self):
        """Validações pós-inicialização"""
        if self.idempotency_key and len(self.idempotency_key) > 255:
            raise ValueError("Idempotency key must be 255 characters or less")
    
    def get_queue_name(self) -> str:
        """Retorna o nome da fila baseado no tipo de tarefa"""
        return f"{self.task_type}_queue"
    
    def get_priority_queue_name(self) -> str:
        """Retorna o nome da fila com prioridade"""
        return f"{self.metadata.priority.name.lower()}_{self.get_queue_name()}"
    
    def to_dict(self) -> Dict[str, Any]:
        """Serializa a tarefa para dicionário"""
        return {
            "metadata": self.metadata.to_dict(),
            "task_type": self.task_type,
            "payload": self.payload,
            "idempotency_key": self.idempotency_key,
            "max_retries": self.max_retries,
            "retry_delay": self.retry_delay
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> 'BaseTask':
        """Cria uma tarefa a partir de dicionário"""
        metadata_dict = data["metadata"]
        
        # Converte strings de volta para objetos
        metadata_dict["task_id"] = UUID(metadata_dict["task_id"])
        metadata_dict["created_at"] = datetime.fromisoformat(metadata_dict["created_at"])
        
        if metadata_dict.get("scheduled_for"):
            metadata_dict["scheduled_for"] = datetime.fromisoformat(metadata_dict["scheduled_for"])
        
        metadata_dict["priority"] = TaskPriority(metadata_dict["priority"])
        
        if metadata_dict.get("parent_task_id"):
            metadata_dict["parent_task_id"] = UUID(metadata_dict["parent_task_id"])
        
        if metadata_dict.get("expires_at"):
            metadata_dict["expires_at"] = datetime.fromisoformat(metadata_dict["expires_at"])
        
        metadata = TaskMetadata(**metadata_dict)
        
        return cls(
            metadata=metadata,
            task_type=data["task_type"],
            payload=data["payload"],
            idempotency_key=data.get("idempotency_key"),
            max_retries=data.get("max_retries", 3),
            retry_delay=data.get("retry_delay", 60)
        )

@dataclass(frozen=True)
class PaymentWebhookTask(BaseTask):
    """Tarefa para processar webhook de pagamento"""
    
    def __init__(self, provider: str, event_data: Dict[str, Any], **kwargs):
        payload = {
            "provider": provider,
            "event_data": event_data,
            "received_at": datetime.utcnow().isoformat()
        }
        
        # Define metadados padrão para webhooks
        metadata = kwargs.get("metadata")
        if not metadata:
            metadata = TaskMetadata(
                tenant_id=kwargs.pop("tenant_id"),
                priority=TaskPriority.CRITICAL,
                source=f"webhook_{provider}"
            )
        
        super().__init__(
            metadata=metadata,
            task_type="payment_webhook",
            payload=payload,
            idempotency_key=event_data.get("idempotency_key") or event_data.get("event_id"),
            **kwargs
        )

class TaskValidator:
    """Validador de tarefas"""
    
    @staticmethod
    def validate_task(task: BaseTask) -> List[str]:
        """Valida uma tarefa e retorna lista de erros"""
        errors = []
        
        # Valida metadados
        if not task.metadata.tenant_id:
            errors.append("tenant_id é obrigatório")
        
        if task.metadata.expires_at and task.metadata.expires_at < datetime.utcnow():
            errors.append("task já expirou")
        
        # Valida payload baseado no tipo
        if task.task_type == "payment_webhook":
            if "provider" not in task.payload:
                errors.append("provider é obrigatório para webhooks")
            if "event_data" not in task.payload:
                errors.append("event_data é obrigatório para webhooks")
        
        elif task.task_type == "entitlement_grant":
            required = ["user_id", "entitlement_type", "scope"]
            for field in required:
                if field not in task.payload:
                    errors.append(f"{field} é obrigatório para grant de acesso")
        
        # Valida idempotency key
        if task.idempotency_key and len(task.idempotency_key) > 255:
            errors.append("idempotency_key deve ter no máximo 255 caracteres")
        
        # Valida retries
        if task.max_retries < 0 or task.max_retries > 10:
            errors.append("max_retries deve estar entre 0 e 10")
        
        if task.retry_delay < 0 or task.retry_delay > 3600:
            errors.append("retry_delay deve estar entre 0 e 3600 segundos")
        
        return errors
    
    @staticmethod
    def is_valid(task: BaseTask) -> bool:
        """Verifica se a tarefa é válida"""
        return len(TaskValidator.validate_task(task)) == 0


# Mapeamento de tipos de tarefa para classes
TASK_TYPE_MAP = {
    "payment_webhook": PaymentWebhookTask,
    "payment_intent_update": PaymentIntentUpdateTask,
    "invoice_generation": InvoiceGenerationTask,
    "subscription_renewal": SubscriptionRenewalTask,
    "entitlement_grant": EntitlementGrantTask,
    "notification_send": NotificationSendTask,
    "antifraud_analysis": AntifraudAnalysisTask,
    "tracking_event": TrackingEventTask,
    "metrics_collection": MetricsCollectionTask,
    "audit_log": AuditLogTask,
    "cleanup": CleanupTask,
}


def deserialize_task(data: Dict[str, Any]) -> BaseTask:
    """Desserializa uma tarefa a partir de dicionário"""
    task_type = data.get("task_type")
    
    if task_type not in TASK_TYPE_MAP:
        # Usa BaseTask como fallback
        return BaseTask.from_dict(data)
    
    # Desserializa usando a classe específica
    task_class = TASK_TYPE_MAP[task_type]
    
    # Extrai parâmetros específicos do payload
    payload = data["payload"]
    metadata_dict = data["metadata"]
    
    # Converte strings de volta para objetos
    metadata_dict["task_id"] = UUID(metadata_dict["task_id"])
    metadata_dict["created_at"] = datetime.fromisoformat(metadata_dict["created_at"])
    
    if metadata_dict.get("scheduled_for"):
        metadata_dict["scheduled_for"] = datetime.fromisoformat(metadata_dict["scheduled_for"])
    
    metadata_dict["priority"] = TaskPriority(metadata_dict["priority"])
    
    metadata = TaskMetadata(**metadata_dict)
    
    # Cria instância da tarefa específica
    if task_type == "payment_webhook":
        return PaymentWebhookTask(
            metadata=metadata,
            provider=payload["provider"],
            event_data=payload["event_data"],
            idempotency_key=data.get("idempotency_key"),
            max_retries=data.get("max_retries", 3),
            retry_delay=data.get("retry_delay", 60)
        )
    
    elif task_type == "entitlement_grant":
        from ..models.entitlement import EntitlementScope
        return EntitlementGrantTask(
            metadata=metadata,
            user_id=payload["user_id"],
            entitlement_type=payload["entitlement_type"],
            scope=EntitlementScope(payload["scope"]),
            idempotency_key=data.get("idempotency_key"),
            max_retries=data.get("max_retries", 3),
            retry_delay=data.get("retry_delay", 60)
        )
    
    # Para outros tipos, usa a lógica genérica
    return task_class(
        metadata=metadata,
        task_type=task_type,
        payload=payload,
        idempotency_key=data.get("idempotency_key"),
        max_retries=data.get("max_retries", 3),
        retry_delay=data.get("retry_delay", 60)
    )
