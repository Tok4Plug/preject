# app/providers/payment/sepa_instant.py
from app.providers.payment.base import PaymentProviderRequest, PaymentProviderResponse


class SepaInstantProvider:
    name = "sepa_instant"
    supported_rails = ("SEPA_INSTANT",)

    async def create_payment(self, request: PaymentProviderRequest) -> PaymentProviderResponse:
        return PaymentProviderResponse(self.name, f"sep_{request.intent_id}", "awaiting_payment")

    async def validate_webhook(self, payload: bytes, signature: str) -> bool:
        return bool(signature)
