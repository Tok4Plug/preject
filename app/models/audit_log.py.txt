# app/models/audit_log.py
"""
Modelo de AuditLog - Registro imutável de eventos para compliance e observabilidade.
Fonte da verdade para auditoria, forense e debugging.
"""

import uuid
from datetime import datetime
from typing import Optional, Dict, Any, List
from sqlalchemy import Column, String, DateTime, Boolean, JSON, Integer, Float, ForeignKey, Index, Text
from sqlalchemy.orm import relationship, Mapped, mapped_column
from sqlalchemy.dialects.postgresql import UUID, JSONB
import enum

from app.database import Base

class AuditResourceType(enum.Enum):
    """Tipos de recursos auditáveis."""
    USER = "user"
    TENANT = "tenant"
    BOT = "bot"
    OFFER = "offer"
    PAYMENT_INTENT = "payment_intent"
    INVOICE = "invoice"
    SUBSCRIPTION = "subscription"
    ENTITLEMENT = "entitlement"
    WEBHOOK = "webhook"
    AUDIT_LOG = "audit_log"
    SYSTEM = "system"


class AuditLog(Base):
    """
    Log de auditoria - registro imutável de eventos.
    
    Princípios:
    1. Imutabilidade: Nunca pode ser alterado ou deletado
    2. Completo: Contém todos os dados necessários para replay
    3. Isolamento por tenant
    4. Indexado para buscas eficientes
    5. Retenção configurável (para compliance)
    """
    
    __tablename__ = "audit_logs"
    
    # Identificação
    id: Mapped[str] = mapped_column(String(36), primary_key=True, default=lambda: str(uuid.uuid4()))
    tenant_id: Mapped[str] = mapped_column(String(36), ForeignKey("saas.tenants.id"), nullable=False, index=True)
    
    # Ação
    action: Mapped[AuditAction] = mapped_column(String(100), nullable=False, index=True)
    action_timestamp: Mapped[datetime] = mapped_column(DateTime, nullable=False, default=datetime.utcnow, index=True)
    
    # Ator (quem realizou a ação)
    actor_id: Mapped[Optional[str]] = mapped_column(String(100), index=True)  # user_id, system, admin_id, etc.
    actor_type: Mapped[Optional[str]] = mapped_column(String(50), index=True)  # user, system, admin, api
    actor_ip: Mapped[Optional[str]] = mapped_column(String(50))
    actor_user_agent: Mapped[Optional[str]] = mapped_column(Text)
    
    # Recurso afetado
    resource_type: Mapped[AuditResourceType] = mapped_column(String(50), nullable=False, index=True)
    resource_id: Mapped[Optional[str]] = mapped_column(String(100), index=True)
    resource_previous_state: Mapped[Optional[Dict[str, Any]]] = mapped_column(JSONB)  # Estado anterior (para updates)
    resource_new_state: Mapped[Optional[Dict[str, Any]]] = mapped_column(JSONB)  # Novo estado (para updates)
    
    # Contexto e detalhes
    details: Mapped[Dict[str, Any]] = mapped_column(JSONB, nullable=False, default=dict)
    metadata: Mapped[Dict[str, Any]] = mapped_column(JSONB, nullable=False, default=dict)
    
    # Rastreamento
    correlation_id: Mapped[Optional[str]] = mapped_column(String(100), index=True)  # Para rastrear operações entre sistemas
    request_id: Mapped[Optional[str]] = mapped_column(String(100), index=True)  # ID da requisição HTTP
    session_id: Mapped[Optional[str]] = mapped_column(String(100), index=True)
    
    # Status da ação
    success: Mapped[bool] = mapped_column(Boolean, nullable=False, default=True)
    error_message: Mapped[Optional[str]] = mapped_column(Text)
    error_code: Mapped[Optional[str]] = mapped_column(String(100))
    
    # Localização
    country_code: Mapped[Optional[str]] = mapped_column(String(10))
    region: Mapped[Optional[str]] = mapped_column(String(50))
    
    # Timestamps (o próprio log tem timestamp, mas também pode ter o timestamp do evento)
    created_at: Mapped[datetime] = mapped_column(DateTime, nullable=False, default=datetime.utcnow, index=True)
    
    # Relacionamentos
    tenant: Mapped["Tenant"] = relationship(back_populates="audit_logs")
    user: Mapped[Optional["User"]] = relationship(back_populates="audit_logs")
    bot: Mapped[Optional["Bot"]] = relationship(back_populates="audit_logs")
    offer: Mapped[Optional["Offer"]] = relationship(back_populates="audit_logs")
    payment_intent: Mapped[Optional["PaymentIntent"]] = relationship(back_populates="audit_logs")
    invoice: Mapped[Optional["Invoice"]] = relationship(back_populates="audit_logs")
    subscription: Mapped[Optional["Subscription"]] = relationship(back_populates="audit_logs")
    entitlement: Mapped[Optional["Entitlement"]] = relationship(back_populates="audit_logs")
    
    # Índices compostos para buscas comuns
    __table_args__ = (
        Index('ix_audit_logs_tenant_action_time', 'tenant_id', 'action', 'action_timestamp'),
        Index('ix_audit_logs_resource', 'resource_type', 'resource_id', 'action_timestamp'),
        Index('ix_audit_logs_actor', 'actor_type', 'actor_id', 'action_timestamp'),
        Index('ix_audit_logs_correlation', 'correlation_id'),
        Index('ix_audit_logs_created', 'created_at'),
        {'schema': 'saas'}
    )
    
    def __repr__(self):
        return f"<AuditLog(id='{self.id}', tenant='{self.tenant_id}', action='{self.action}', resource='{self.resource_type}:{self.resource_id}')>"
    
    def to_dict(self, include_resources: bool = False) -> Dict[str, Any]:
        """Converte o modelo para dicionário."""
        data = {
            "id": self.id,
            "tenant_id": self.tenant_id,
            "action": self.action.value if isinstance(self.action, enum.Enum) else self.action,
            "action_timestamp": self.action_timestamp.isoformat() if self.action_timestamp else None,
            "actor_id": self.actor_id,
            "actor_type": self.actor_type,
            "actor_ip": self.actor_ip,
            "actor_user_agent": self.actor_user_agent,
            "resource_type": self.resource_type.value if isinstance(self.resource_type, enum.Enum) else self.resource_type,
            "resource_id": self.resource_id,
            "details": self.details,
            "metadata": self.metadata,
            "correlation_id": self.correlation_id,
            "request_id": self.request_id,
            "session_id": self.session_id,
            "success": self.success,
            "error_message": self.error_message,
            "error_code": self.error_code,
            "country_code": self.country_code,
            "region": self.region,
            "created_at": self.created_at.isoformat() if self.created_at else None
        }
        
        if include_resources:
            if self.resource_previous_state:
                data["resource_previous_state"] = self.resource_previous_state
            if self.resource_new_state:
                data["resource_new_state"] = self.resource_new_state
        
        return data
    
    @classmethod
    def create_log(
        cls,
        tenant_id: str,
        action: AuditAction,
        actor_id: Optional[str] = None,
        actor_type: Optional[str] = None,
        resource_type: AuditResourceType = AuditResourceType.SYSTEM,
        resource_id: Optional[str] = None,
        resource_previous_state: Optional[Dict[str, Any]] = None,
        resource_new_state: Optional[Dict[str, Any]] = None,
        details: Optional[Dict[str, Any]] = None,
        metadata: Optional[Dict[str, Any]] = None,
        correlation_id: Optional[str] = None,
        request_id: Optional[str] = None,
        session_id: Optional[str] = None,
        success: bool = True,
        error_message: Optional[str] = None,
        error_code: Optional[str] = None,
        actor_ip: Optional[str] = None,
        actor_user_agent: Optional[str] = None,
        country_code: Optional[str] = None,
        region: Optional[str] = None,
        action_timestamp: Optional[datetime] = None
    ) -> "AuditLog":
        """Método factory para criar logs de auditoria."""
        return cls(
            tenant_id=tenant_id,
            action=action,
            actor_id=actor_id,
            actor_type=actor_type,
            resource_type=resource_type,
            resource_id=resource_id,
            resource_previous_state=resource_previous_state,
            resource_new_state=resource_new_state,
            details=details or {},
            metadata=metadata or {},
            correlation_id=correlation_id,
            request_id=request_id,
            session_id=session_id,
            success=success,
            error_message=error_message,
            error_code=error_code,
            actor_ip=actor_ip,
            actor_user_agent=actor_user_agent,
            country_code=country_code,
            region=region,
            action_timestamp=action_timestamp or datetime.utcnow(),
            created_at=datetime.utcnow()
        )
    
    @property
    def is_system_event(self) -> bool:
        """Verifica se é um evento do sistema (sem ator específico)."""
        return self.actor_type == "system" or self.actor_id == "system"
    
    @property
    def is_user_event(self) -> bool:
        """Verifica se é um evento de usuário."""
        return self.actor_type == "user"
    
    @property
    def is_admin_event(self) -> bool:
        """Verifica se é um evento administrativo."""
        return self.actor_type == "admin"
    
    @property
    def has_resource_changes(self) -> bool:
        """Verifica se o log contém mudanças de estado do recurso."""
        return bool(self.resource_previous_state or self.resource_new_state)
    
    def get_resource_diff(self) -> Dict[str, Any]:
        """Retorna as diferenças entre o estado anterior e o novo."""
        if not self.has_resource_changes:
            return {}
        
        old = self.resource_previous_state or {}
        new = self.resource_new_state or {}
        
        diff = {}
        
        # Encontra campos adicionados ou modificados
        for key in set(old.keys()) | set(new.keys()):
            old_val = old.get(key)
            new_val = new.get(key)
            
            if old_val != new_val:
                diff[key] = {
                    "old": old_val,
                    "new": new_val
                }
        
        return diff


class AuditRetentionPolicy(Base):
    """
    Política de retenção de logs de auditoria.
    Configuração por tenant para compliance.
    """
    
    __tablename__ = "audit_retention_policies"
    
    id: Mapped[str] = mapped_column(String(36), primary_key=True, default=lambda: str(uuid.uuid4()))
    tenant_id: Mapped[str] = mapped_column(String(36), ForeignKey("saas.tenants.id"), nullable=False, unique=True, index=True)
    
    # Retenção por tipo de ação
    retention_days_default: Mapped[int] = mapped_column(Integer, nullable=False, default=365)  # Dias padrão
    retention_days_financial: Mapped[int] = mapped_column(Integer, nullable=False, default=1825)  # 5 anos para financeiro
    retention_days_user_data: Mapped[int] = mapped_column(Integer, nullable=False, default=1095)  # 3 anos para dados de usuário
    retention_days_system: Mapped[int] = mapped_column(Integer, nullable=False, default=90)  # 90 dias para logs de sistema
    
    # Configurações de arquivamento
    archive_after_days: Mapped[int] = mapped_column(Integer, nullable=False, default=90)  # Arquiva após X dias
    archive_to_cold_storage: Mapped[bool] = mapped_column(Boolean, nullable=False, default=False)
    
    # Configurações de compliance
    compliance_standard: Mapped[Optional[str]] = mapped_column(String(100))  # GDPR, LGPD, HIPAA, etc.
    require_immutable_logs: Mapped[bool] = mapped_column(Boolean, nullable=False, default=True)
    
    # Metadata
    metadata: Mapped[Dict[str, Any]] = mapped_column(JSONB, nullable=False, default=dict)
    
    # Timestamps
    created_at: Mapped[datetime] = mapped_column(DateTime, nullable=False, default=datetime.utcnow)
    updated_at: Mapped[datetime] = mapped_column(DateTime, nullable=False, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # Relacionamentos
    tenant: Mapped["Tenant"] = relationship()
    
    def __repr__(self):
        return f"<AuditRetentionPolicy(tenant_id='{self.tenant_id}', default_days={self.retention_days_default})>"
    
    def get_retention_days_for_action(self, action: AuditAction) -> int:
        """Retorna dias de retenção para uma ação específica."""
        # Mapeamento de ações para categorias de retenção
        financial_actions = {
            AuditAction.PAYMENT_INTENT_CREATED,
            AuditAction.PAYMENT_CONFIRMED,
            AuditAction.PAYMENT_REFUNDED,
            AuditAction.PAYMENT_FAILED,
            AuditAction.SUBSCRIPTION_CREATED,
            AuditAction.SUBSCRIPTION_RENEWED,
            AuditAction.SUBSCRIPTION_CANCELLED
        }
        
        user_data_actions = {
            AuditAction.USER_CREATED,
            AuditAction.USER_UPDATED,
            AuditAction.USER_DELETED,
            AuditAction.USER_SUSPENDED,
            AuditAction.USER_LOGIN,
            AuditAction.USER_LOGOUT
        }
        
        if action in financial_actions:
            return self.retention_days_financial
        elif action in user_data_actions:
            return self.retention_days_user_data
        else:
            return self.retention_days_default
    
    def should_archive(self, log_age_days: int) -> bool:
        """Verifica se um log deve ser arquivado baseado na idade."""
        return log_age_days > self.archive_after_days
    
    def should_delete(self, log_age_days: int, action: AuditAction) -> bool:
        """Verifica se um log deve ser deletado baseado na idade e ação."""
        retention_days = self.get_retention_days_for_action(action)
        return log_age_days > retention_days
