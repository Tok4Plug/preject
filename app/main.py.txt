+def _load_optional(module_path: str, attr_name: str) -> Optional[Any]:
+    """Carrega dependência opcional sem quebrar inicialização da API."""
+    try:
+        module = importlib.import_module(module_path)
+        return getattr(module, attr_name, None)
+    except Exception as exc:  # pragma: no cover - proteção para ambiente parcial
+        logger.warning(
+            "optional_dependency_unavailable",
+            module=module_path,
+            attribute=attr_name,
+            error=str(exc),
+        )
+        return None
